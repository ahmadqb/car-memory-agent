"""python test_memory.py  (plain asserts, no pytest needed)"""
import os
from datetime import datetime, timedelta

os.environ["MEMORY_DB"] = "test_memory.db"
import importlib
import memory
importlib.reload(memory)
from signals import generate_events

U = "t"


def fresh():
    if os.path.exists("test_memory.db"):
        os.remove("test_memory.db")
    memory.init_db(reset=True)


def facts():
    return {f["key"]: f for f in memory.get_profile(U)}


# 1. learns the main preferences from simulated days
fresh()
for ts, s, v in generate_events():
    memory.add_observation(U, s, v, ts, derive=False)
memory.derive_facts(U)
f = facts()
assert f["cabin_temp_set@morning"]["value"] == "22"
assert f["cabin_temp_set@evening"]["value"] == "21"
assert f["media_genre_played@morning"]["value"] == "electronic"
assert f["media_genre_played@evening"]["value"] == "jazz"
print("1 ok: learns preferences")

# 2. too little data -> no fact
fresh()
t0 = datetime(2026, 10, 1, 7, 0)
memory.add_observation(U, "cabin_temp_set", "22", t0)
memory.add_observation(U, "cabin_temp_set", "22", t0 + timedelta(days=1))
assert facts() == {}
print("2 ok: no fact with 2 observations")

# 3. mixed signals -> no fact (no clear preference)
fresh()
for i, v in enumerate(["20", "21", "22", "23", "24", "20"]):
    memory.add_observation(U, "cabin_temp_set", v, t0 + timedelta(days=i))
assert "cabin_temp_set@morning" not in facts()
print("3 ok: no fact without a clear majority")

# 4. preference changes: 10 new observations of 23 replace the old 22
fresh()
for i in range(10):
    memory.add_observation(U, "cabin_temp_set", "22", t0 + timedelta(days=i))
assert facts()["cabin_temp_set@morning"]["value"] == "22"
for i in range(10, 20):
    memory.add_observation(U, "cabin_temp_set", "23", t0 + timedelta(days=i))
assert facts()["cabin_temp_set@morning"]["value"] == "23"
print("4 ok: preference change is picked up")

# 5. forget one key, and it does not come back
fresh()
for i in range(5):
    memory.add_observation(U, "cabin_temp_set", "22", t0 + timedelta(days=i))
memory.add_observation(U, "media_genre_played", "jazz", t0)
assert memory.forget(U, "cabin_temp_set@morning") == 1
memory.derive_facts(U)
assert "cabin_temp_set@morning" not in facts()
print("5 ok: forget one key")

# 6. forget everything
memory.forget(U)
assert facts() == {}
print("6 ok: forget all")

os.remove("test_memory.db")
print("ALL TESTS PASSED")

"""python test_client.py -> starts car_server.py, lists its tools and calls a few of them."""
import asyncio
import sys

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client


async def main():
    params = StdioServerParameters(command=sys.executable, args=["car_server.py"])
    async with stdio_client(params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()

            tools = await session.list_tools()
            print("Tools exposed by the car:")
            for t in tools.tools:
                print(f"  - {t.name}: {t.description}")

            for name, args in [
                ("get_cabin_temperature", {}),
                ("set_cabin_temperature", {"temp_c": 22}),
                ("get_cabin_temperature", {}),
                ("set_cabin_temperature", {"temp_c": 40}),   # must be refused
                ("play_media", {"genre": "electronic"}),
                ("get_battery_level", {}),
            ]:
                result = await session.call_tool(name, args)
                print(f"{name}({args}) -> {result.content[0].text}")


if __name__ == "__main__":
    asyncio.run(main())

signals.py:

"""Simulated vehicle/user signals for one driver over several days."""
import random
from datetime import datetime, timedelta


def _pick(rng, main, other, p_main):
    return main if rng.random() < p_main else other


def generate_events(days: int = 14, seed: int = 42, start: datetime = datetime(2026, 10, 1)):
    """Returns a list of (timestamp, signal, value). The driver likes 22C and electronic
    music in the morning and 21C and jazz in the evening, but is not 100% consistent."""
    rng = random.Random(seed)
    events = []
    for d in range(days):
        day = start + timedelta(days=d)

        t = day.replace(hour=7, minute=rng.randint(0, 59))
        events.append((t, "cabin_temp_set", _pick(rng, "22", "21", 0.85)))
        events.append((t + timedelta(minutes=1), "media_genre_played", _pick(rng, "electronic", "pop", 0.8)))
        events.append((t + timedelta(minutes=2), "destination_set", _pick(rng, "office", "gym", 0.9)))

        t = day.replace(hour=18, minute=rng.randint(0, 59))
        events.append((t, "cabin_temp_set", _pick(rng, "21", "22", 0.8)))
        events.append((t + timedelta(minutes=1), "media_genre_played", _pick(rng, "jazz", "pop", 0.75)))
        events.append((t + timedelta(minutes=2), "destination_set", _pick(rng, "home", "supermarket", 0.95)))
    return events

memory.py:

"""Driver memory: raw observations -> derived facts (with confidence), view, forget."""
import os
import sqlite3
from collections import Counter
from datetime import datetime

DB_PATH = os.getenv("MEMORY_DB", os.path.join(os.path.dirname(os.path.abspath(__file__)), "memory.db"))

WINDOW = 10          # only the last N observations per (signal, time bucket) count -> preferences can change
MIN_SUPPORT = 3      # need at least this many observations before we claim to "know" something
MIN_CONFIDENCE = 0.6 # share of the window that must agree on one value


def connect():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(reset: bool = False):
    conn = connect()
    if reset:
        conn.executescript("DROP TABLE IF EXISTS observations; DROP TABLE IF EXISTS facts;")
    conn.executescript("""
        CREATE TABLE IF NOT EXISTS observations (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id TEXT NOT NULL,
            signal TEXT NOT NULL,
            bucket TEXT NOT NULL,
            value TEXT NOT NULL,
            ts TEXT NOT NULL
        );
        CREATE TABLE IF NOT EXISTS facts (
            user_id TEXT NOT NULL,
            key TEXT NOT NULL,
            value TEXT NOT NULL,
            confidence REAL NOT NULL,
            support INTEGER NOT NULL,
            updated_at TEXT NOT NULL,
            PRIMARY KEY (user_id, key)
        );
    """)
    conn.commit()
    conn.close()


def time_bucket(ts: datetime) -> str:
    h = ts.hour
    if 5 <= h < 11:
        return "morning"
    if 11 <= h < 17:
        return "afternoon"
    if 17 <= h < 22:
        return "evening"
    return "night"


def add_observation(user_id: str, signal: str, value: str, ts: datetime | None = None, derive: bool = True):
    """Store one vehicle/user event. Only USER-originated events belong here,
    not actions the assistant did itself (otherwise it would learn from its own behavior)."""
    ts = ts or datetime.now()
    conn = connect()
    conn.execute(
        "INSERT INTO observations (user_id, signal, bucket, value, ts) VALUES (?, ?, ?, ?, ?)",
        (user_id, signal, time_bucket(ts), str(value), ts.isoformat()),
    )
    conn.commit()
    conn.close()
    if derive:
        derive_facts(user_id)


def derive_facts(user_id: str) -> list[dict]:
    """Recompute facts from the most recent observations of every (signal, bucket)."""
    conn = connect()
    groups = conn.execute(
        "SELECT DISTINCT signal, bucket FROM observations WHERE user_id = ?", (user_id,)
    ).fetchall()
    now = datetime.now().isoformat()
    for g in groups:
        rows = conn.execute(
            "SELECT value FROM observations WHERE user_id=? AND signal=? AND bucket=? "
            "ORDER BY ts DESC LIMIT ?",
            (user_id, g["signal"], g["bucket"], WINDOW),
        ).fetchall()
        values = [r["value"] for r in rows]
        top_value, top_count = Counter(values).most_common(1)[0]
        confidence = top_count / len(values)
        key = f"{g['signal']}@{g['bucket']}"
        if len(values) >= MIN_SUPPORT and confidence >= MIN_CONFIDENCE:
            conn.execute(
                "INSERT INTO facts (user_id, key, value, confidence, support, updated_at) "
                "VALUES (?, ?, ?, ?, ?, ?) "
                "ON CONFLICT(user_id, key) DO UPDATE SET value=excluded.value, "
                "confidence=excluded.confidence, support=excluded.support, updated_at=excluded.updated_at",
                (user_id, key, top_value, round(confidence, 2), len(values), now),
            )
        else:
            conn.execute("DELETE FROM facts WHERE user_id=? AND key=?", (user_id, key))
    conn.commit()
    conn.close()
    return get_profile(user_id)


def get_profile(user_id: str) -> list[dict]:
    conn = connect()
    rows = conn.execute(
        "SELECT key, value, confidence, support, updated_at FROM facts "
        "WHERE user_id=? ORDER BY key", (user_id,)
    ).fetchall()
    conn.close()
    return [dict(r) for r in rows]


def forget(user_id: str, key: str | None = None) -> int:
    """GDPR-style deletion. key=None deletes everything about the user.
    Raw observations are deleted too, otherwise the fact would be re-derived."""
    conn = connect()
    if key is None:
        n = conn.execute("DELETE FROM facts WHERE user_id=?", (user_id,)).rowcount
        conn.execute("DELETE FROM observations WHERE user_id=?", (user_id,))
    else:
        signal, _, bucket = key.partition("@")
        n = conn.execute("DELETE FROM facts WHERE user_id=? AND key=?", (user_id, key)).rowcount
        conn.execute(
            "DELETE FROM observations WHERE user_id=? AND signal=? AND bucket=?",
            (user_id, signal, bucket),
        )
    conn.commit()
    conn.close()
    return n


car_server.py:

"""MCP server that plays 'the car'. Run by an MCP client over stdio (see test_client.py)."""
from mcp.server.fastmcp import FastMCP

mcp = FastMCP("car")

# Mock vehicle state (lives as long as the server process)
STATE = {"cabin_temp_c": 20.0, "battery_pct": 76, "now_playing": None, "location": "Munich"}

MIN_TEMP, MAX_TEMP = 16.0, 28.0


@mcp.tool()
def get_cabin_temperature() -> float:
    """Current cabin temperature setpoint in degrees Celsius."""
    return STATE["cabin_temp_c"]


@mcp.tool()
def set_cabin_temperature(temp_c: float) -> str:
    """Set the cabin temperature in degrees Celsius (allowed range 16 to 28)."""
    if not MIN_TEMP <= temp_c <= MAX_TEMP:
        return f"Refused: {temp_c} C is outside the allowed range {MIN_TEMP}-{MAX_TEMP} C."
    STATE["cabin_temp_c"] = temp_c
    return f"Cabin temperature set to {temp_c} C."


@mcp.tool()
def get_battery_level() -> int:
    """Battery state of charge in percent."""
    return STATE["battery_pct"]


@mcp.tool()
def play_media(genre: str) -> str:
    """Start playing a playlist of the given genre, e.g. 'jazz' or 'electronic'."""
    STATE["now_playing"] = genre
    return f"Now playing: {genre} playlist."


@mcp.tool()
def get_current_location() -> str:
    """Current location of the vehicle (mock data)."""
    return STATE["location"]


if __name__ == "__main__":
    mcp.run(transport="stdio")
